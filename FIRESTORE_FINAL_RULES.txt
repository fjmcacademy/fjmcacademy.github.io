rules_version = '2';

service cloud.firestore {

  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null
        && request.auth.token.email == "fjmcacademy1008@gmail.com";
    }

    // =====================================================
    // USERS
    // =====================================================
    match /users/{userId} {

      // Student can access only own user document.
      // Admin can read user documents so the Admin panel
      // can display device reservations.
      allow read: if isAdmin()
                  || (
                    request.auth != null
                    && request.auth.uid == userId
                  );

      // Student can write only own user document.
      allow create, update, delete:
        if request.auth != null
        && request.auth.uid == userId;

      // ===================================================
      // OLD DEVICE COLLECTION
      // ===================================================
      match /devices/{deviceId} {

        allow read, create, update, delete:
          if isAdmin()
          || (
            request.auth != null
            && request.auth.uid == userId
          );
      }

      // ===================================================
      // FINAL DEVICE SLOT SYSTEM
      // users/{uid}/deviceSlots/main
      //
      // mobileDeviceId  = Phone OR Tablet
      // desktopDeviceId = Laptop OR Desktop
      // ===================================================
      match /deviceSlots/{slotId} {

        allow read, create, update:
          if isAdmin()
          || (
            request.auth != null
            && request.auth.uid == userId
          );

        allow delete:
          if isAdmin();
      }
    }

    // =====================================================
    // STUDENTS
    // =====================================================
    match /students/{studentId} {

      allow read: if isAdmin()
                  || (
                    request.auth != null
                    && request.auth.token.email == resource.data.email
                  );

      allow create, update, delete:
        if isAdmin();
    }

    // =====================================================
    // COURSES
    // =====================================================
    match /courses/{courseId} {

      allow read:
        if request.auth != null;

      allow create, update, delete:
        if isAdmin();
    }

    // =====================================================
    // TESTS
    // =====================================================
    match /tests/{testId} {

      allow read:
        if request.auth != null;

      allow write:
        if isAdmin();
    }

    // =====================================================
    // TEST RESULTS
    // =====================================================
    match /testResults/{resultId} {

      allow read, create, update:
        if request.auth != null;

      allow delete:
        if isAdmin();
    }

    // =====================================================
    // DEFAULT DENY
    // =====================================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
